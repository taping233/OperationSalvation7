      if (barTone === tone) return;         // 只在提示阶段改变时更新，不每 400ms 重建
      removeBar();
      barTone = tone;
      var el = document.createElement('div');
      el.textContent = text;
      el.style.cssText = 'position:fixed;left:50%;bottom:18px;transform:translateX(-50%);z-index:9999;'
        + 'padding:8px 18px;border-radius:6px;'
        + (tone === 'fail'
            ? 'background:#3a1216;color:#ffd9d4;border:1px solid #a4524a;'
            : 'background:#12262f;color:#b2c4c9;border:1px solid #48667455;')
        + 'font:13px/1.6 system-ui,sans-serif;box-shadow:0 6px 18px rgba(0,0,0,.5)';
      if (closable) {
        var close = document.createElement('button');
        close.textContent = '×';
        close.setAttribute('aria-label', '收起启动提示');
        close.style.cssText = 'margin-left:12px;background:none;border:0;color:inherit;font:inherit;cursor:pointer';
        close.addEventListener('click', dismiss);
        el.appendChild(close);
      }
      document.body.appendChild(el);
      bar = el;
    }
    function onResourceError(e) {
      if (e && e.target && (e.target.tagName === 'SCRIPT' || e.target.tagName === 'LINK')) {
        stopTimer();                       // 已证实失败，不再用计时或就绪覆盖它
        showBar('资源加载失败——请刷新页面重试', 'fail', true);
      }
    }
    window.addEventListener('error', onResourceError, true);
    window.addEventListener('pagehide', dismiss);
    guardTimer = setInterval(function () {
      var sdt = window.SDT;
      var title = document.getElementById('title');
      if (sdt && sdt.game && sdt.game.state && sdt.game.state !== 'boot'
          && title && title.dataset.controlsBound === 'true') {
        removeBar();                       // 晚就绪也撤条停表，真实资源错误仍可报告
        stopTimer();
        return;
      }
      var elapsed = Date.now() - started;
      if (elapsed >= 12000) {
        showBar('加载耗时较长——仍在等待启动完成；可继续稍候，或刷新页面重试', 'slow', true);
      } else if (elapsed >= 3000) {
        showBar('资源加载中……首次启动需要预热美术素材，请稍候', 'wait', false);
      }
    }, 400);
  })();
</script>
