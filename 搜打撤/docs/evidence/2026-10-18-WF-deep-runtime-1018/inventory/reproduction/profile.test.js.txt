let current=null, depth=0;
const frames=new Map();let nextFrame=0;
async function flushFrames() {
  for(let i=0;i<12;i++) {
    await Promise.resolve();await Promise.resolve();
    if(!frames.size) return;
    const callbacks=[...frames.values()];frames.clear();
    for(const callback of callbacks)callback(performance.now());
  }
  expect(frames.size,'queued frame work must settle').toBe(0);
}
function instrument(obj,name,label=name) {
  const original=obj[name]; if(typeof original!=='function') return;
  obj[name]=function(...args) {
    if(!current) return original.apply(this,args);
    const rec=current.breakdown[label] ||= {calls:0,ms:0,selfMs:0}; rec.calls++;
    const start=performance.now(), parent=current.child;
    current.child=0; depth++;
    try { return original.apply(this,args); }
    finally { depth--; const elapsed=performance.now()-start, child=current.child; rec.ms+=elapsed;rec.selfMs+=elapsed-child;current.child=parent+elapsed; }
  };
  return ()=>{obj[name]=original;};
}
function quantile(xs,q) {const x=[...xs].sort((a,b)=>a-b);return x[Math.min(x.length-1,Math.ceil(x.length*q)-1)];}
function summarize(samples) {
  const labels=new Set(samples.flatMap(s=>Object.keys(s.breakdown))), breakdown={};
  for(const key of labels) breakdown[key]={calls:quantile(samples.map(s=>s.breakdown[key]?.calls||0),.5),ms:quantile(samples.map(s=>s.breakdown[key]?.ms||0),.5),selfMs:quantile(samples.map(s=>s.breakdown[key]?.selfMs||0),.5)};
  return {n:samples.length,medianMs:quantile(samples.map(s=>s.ms),.5),p95Ms:quantile(samples.map(s=>s.ms),.95),meanCpuMs:samples.reduce((sum,s)=>sum+s.cpuMs,0)/samples.length,medianCpuMs:quantile(samples.map(s=>s.cpuMs),.5),p95CpuMs:quantile(samples.map(s=>s.cpuMs),.95),breakdown};
}
async function measured(fn) {
  const rec={breakdown:{},child:0}, cpu=process.cpuUsage(), start=performance.now();current=rec;
  try {await fn();await flushFrames();}
  finally {current=null;rec.ms=performance.now()-start;const used=process.cpuUsage(cpu);rec.cpuMs=(used.user+used.system)/1000;delete rec.child;}
  return rec;
}
function post(session,method,params={}) {return new Promise((resolve,reject)=>session.post(method,params,(e,r)=>e?reject(e):resolve(r)));}
