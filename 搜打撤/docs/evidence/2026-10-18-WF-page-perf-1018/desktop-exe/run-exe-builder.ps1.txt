$ErrorActionPreference = 'Stop'
$taskDir = $PSScriptRoot
$sourceProject = (Resolve-Path -LiteralPath (Join-Path $taskDir '../..')).Path
$desktopProject = Join-Path $taskDir 'exe-build/project/desktop-app'
$deliveryDir = Join-Path $taskDir 'delivery'
$builderCache = Join-Path $taskDir 'exe-build/builder-cache'
$userBuilderCache = Join-Path $env:USERPROFILE 'AppData/Local/electron-builder/Cache'
$runtimeDir = Join-Path $sourceProject 'desktop-app/node_modules/electron/dist'
$builderCLI = Join-Path $sourceProject 'desktop-app/node_modules/electron-builder/cli.js'
$env:ELECTRON_BUILDER_NSIS_DIR = Join-Path $userBuilderCache 'nsis-3.0.4.1/nsis-3.0.4.1-1mx3n'
$env:ELECTRON_BUILDER_NSIS_RESOURCES_DIR = Join-Path $userBuilderCache 'nsis-resources-3.4.1/nsis-resources-3.4.1-2jx2y'
$env:ELECTRON_BUILDER_7ZIP_PATH = Join-Path $userBuilderCache '7zip@1.0.0/7zip-win-x64-1nrf7/bin/7za.exe'
$env:ELECTRON_BUILDER_CACHE = $builderCache
$env:ELECTRON_SKIP_BINARY_DOWNLOAD = '1'
$env:CSC_IDENTITY_AUTO_DISCOVERY = 'false'
$env:npm_config_offline = 'true'
$env:ELECTRON_BUILDER_BINARIES_MIRROR = ([uri](Join-Path $taskDir 'offline-no-download/')).AbsoluteUri
$env:ELECTRON_MIRROR = $env:ELECTRON_BUILDER_BINARIES_MIRROR
foreach ($required in @($builderCLI, (Join-Path $runtimeDir 'electron.exe'), (Join-Path $env:ELECTRON_BUILDER_NSIS_DIR 'Bin/makensis.exe'), (Join-Path $env:ELECTRON_BUILDER_NSIS_RESOURCES_DIR 'plugins'), $env:ELECTRON_BUILDER_7ZIP_PATH, (Join-Path $desktopProject 'game/index.html'))) {
  if (-not (Test-Path -LiteralPath $required)) { throw "Required offline packaging input missing: $required" }
}
node (Join-Path $taskDir 'check-release-inputs.cjs')
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
New-Item -ItemType Directory -Path $deliveryDir, $builderCache -Force | Out-Null
node $builderCLI --projectDir $desktopProject --win portable --x64 --publish never "--config.electronDist=$runtimeDir" "--config.directories.output=$deliveryDir" --config.npmRebuild=false --config.win.signAndEditExecutable=false 2>&1 | Tee-Object -FilePath (Join-Path $taskDir 'exe-build/build-portable.log')
exit $LASTEXITCODE
