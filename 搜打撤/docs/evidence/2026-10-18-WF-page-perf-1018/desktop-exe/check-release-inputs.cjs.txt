const fs = require('fs');
const path = require('path');
const { spawnSync } = require('child_process');

const sourceRoot = path.resolve(__dirname, '../..');
const snapshot = path.join(__dirname, 'exe-build/project');
const expected = JSON.parse(fs.readFileSync(path.join(snapshot, 'desktop-app/package.json'), 'utf8')).devDependencies.electron.replace(/^[^0-9]*/, '');
const executable = path.join(sourceRoot, 'desktop-app/node_modules/electron/dist/electron.exe');
const probe = spawnSync(executable, ['--version'], { encoding: 'utf8', windowsHide: true });
const actual = (probe.stdout || '').trim().replace(/^v/, '');
if (probe.status !== 0 || actual !== expected) {
  console.error(`Electron runtime mismatch: expected ${expected}, actual ${actual || 'unavailable'}; ${probe.error?.code || probe.status}`);
  process.exit(1);
}
console.log(`Electron runtime matches frozen release input: ${actual}`);
require(path.join(snapshot, 'scripts/check-version.cjs'));
