# 全部从零合成，无任何第三方音频输入。确定性随机种子，可复现。
# 输出: game/assets/sfx/original-full-1027/<profile>.wav (44.1kHz mono int16)

def n(dur):
    return max(1, int(SR * dur))


def t(dur):
    return np.arange(n(dur)) / SR


def exp_env(dur, decay):
    x = t(dur)
    return np.exp(-x * decay)


def adsr(dur, a, d, s_level, r):
    total = n(dur)
    na, nd, nr = n(a), n(d), n(r)
    ns = max(0, total - na - nd - nr)
    env = np.concatenate([
        np.linspace(0, 1, na, endpoint=False),
        np.linspace(1, s_level, nd, endpoint=False),
        np.full(ns, s_level),
        np.linspace(s_level, 0, nr),
    ])[:total]
    return env


def sine(freq, dur):
    return np.sin(2 * np.pi * freq * t(dur))


def glide(f0, f1, dur):
    x = t(dur)
    phase = 2 * np.pi * (f0 * x + (f1 - f0) / (2 * dur) * x * x)
    return np.sin(phase)


def bell(freq, dur, decay=6.0, partials=(1, 2.01, 2.74)):
    x = np.zeros(n(dur))
    for i, p in enumerate(partials):
        x += (0.6 ** i) * sine(freq * p, dur) * exp_env(dur, decay + i * 3)
    return x


def noise(dur, seed):
    rng = np.random.default_rng(seed)
    return rng.standard_normal(n(dur))


def shape(sig, gain_at):
    """gain_at(freq)->gain 的频域塑形。"""
    spec = np.fft.rfft(sig)
    freqs = np.fft.rfftfreq(len(sig), 1 / SR)
    spec *= gain_at(freqs)
    return np.fft.irfft(spec, len(sig))


def lowpass(sig, cutoff):
    return shape(sig, lambda f: 1 / (1 + (f / cutoff) ** 4))


def highpass(sig, cutoff):
    return shape(sig, lambda f: (f / cutoff) ** 2 / (1 + (f / cutoff) ** 2))


def bandpass(sig, lo, hi):
    return shape(sig, lambda f: ((f / lo) ** 2 / (1 + (f / lo) ** 2)) * (1 / (1 + (f / hi) ** 4)))


def sweep_noise(dur, f0, f1, seed, width=1.6):
    """频带随时间从 f0 扫到 f1 的噪声（嗖/风声基底）。"""
    x = noise(dur, seed)
    win = n(0.03)
    hop = win // 2
    out = np.zeros_like(x)
    norm = np.zeros_like(x)
    w = np.hanning(win)
    for start in range(0, len(x) - win, hop):
        pos = start / len(x)
        center = f0 * (f1 / f0) ** pos
        frame = x[start:start + win] * w
        spec = np.fft.rfft(frame)
        freqs = np.fft.rfftfreq(win, 1 / SR)
        g = np.exp(-0.5 * ((freqs - center) / (center * (width - 1) + 60)) ** 2)
        out[start:start + win] += np.fft.irfft(spec * g, win) * w
        norm[start:start + win] += w * w
    return out / np.maximum(norm, 1e-6)


def mix(*sigs):
    total = max(len(s) for s in sigs)
    out = np.zeros(total)
    for s in sigs:
        out[:len(s)] += s
    return out


def at(sig, delay):
    return np.concatenate([np.zeros(n(delay)), sig])


def finish(sig, peak=0.85, fade_in=0.004, fade_out=0.02):
    sig = np.tanh(sig * 1.1)
    m = np.max(np.abs(sig)) or 1.0
    sig = sig / m * peak
    fi, fo = n(fade_in), n(fade_out)
    sig[:fi] *= np.linspace(0, 1, fi)
    sig[-fo:] *= np.linspace(1, 0, fo)
    return sig

def click():
    x = glide(1750, 1150, 0.07) * exp_env(0.07, 55)
    tick = bandpass(noise(0.02, 11), 2500, 6000) * exp_env(0.02, 160)
    return finish(mix(x, at(tick * 0.5, 0)), peak=0.6)

    x += 0.3 * np.sign(glide(233, 196, 0.22)) * exp_env(0.22, 9)
    return finish(lowpass(x, 1400), peak=0.6)

def fanfare(notes, step, hold, decay):
    parts = [at(bell(f, hold, decay), i * step) for i, f in enumerate(notes)]
    return finish(mix(*parts), peak=0.75)

def write_wav(path, sig):
    pcm = np.clip(sig, -1, 1)
    pcm = (pcm * 32767).astype(np.int16)
    with wave.open(path, "wb") as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(SR)
        w.writeframes(pcm.tobytes())


def main():
    os.makedirs(OUT, exist_ok=True)
    manifest = {}
    for name, fn in RECIPES.items():
        sig = fn()
        path = os.path.join(OUT, name + ".wav")
        write_wav(path, sig)
        with open(path, "rb") as f:
            digest = hashlib.sha256(f.read()).hexdigest()
        manifest[name] = {
            "file": "assets/sfx/original-full-1027/" + name + ".wav",
            "duration": round(len(sig) / SR, 3),
            "peak": round(float(np.max(np.abs(sig))), 3),
            "rms": round(float(np.sqrt(np.mean(sig ** 2))), 4),
            "sha256": digest,
        }
        print(f"{name:20s} {len(sig)/SR:5.2f}s  rms={manifest[name]['rms']}")
    mpath = os.path.join(HERE, "manifest.json")
    with open(mpath, "w", encoding="utf-8") as f:
        json.dump(manifest, f, ensure_ascii=False, indent=2, sort_keys=True)
    print(f"\n{len(manifest)} files -> {OUT}")
    print(f"manifest -> {mpath}")


if __name__ == "__main__":
    main()
