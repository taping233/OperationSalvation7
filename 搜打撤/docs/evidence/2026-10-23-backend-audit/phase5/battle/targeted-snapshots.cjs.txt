function value(node, parameters = {}) {
  if (node.type === 'Literal') return node.value;
  if (node.type === 'Identifier' && Object.hasOwn(parameters, node.name)) return parameters[node.name];
  if (node.type === 'UnaryExpression' && node.operator === '-') return -value(node.argument, parameters);
  if (node.type === 'ArrayExpression') return node.elements.map(element => value(element, parameters));
  if (node.type === 'ObjectExpression') {
    const object = {};
    for (const property of node.properties) {
      assert.equal(property.type, 'Property'); assert.equal(property.computed, false); assert.equal(property.kind, 'init'); assert.equal(property.method, false);
      const key = property.key.type === 'Identifier' ? property.key.name : property.key.value;
      assert.equal(Object.hasOwn(object, key), false);
      Object.defineProperty(object, key, { value: value(property.value, parameters), writable: true, enumerable: true, configurable: true });
    }
    return object;
  }
  if (node.type === 'CallExpression' && node.callee.type === 'Identifier' && Object.hasOwn(functions, node.callee.name)) {
    const fn = functions[node.callee.name];
    assert.equal(node.arguments.length, fn.params.length);
    const args = node.arguments.map(argument => value(argument));
    return value(fn.body.body[0].argument, Object.fromEntries(fn.params.map((parameter, index) => [parameter.name, args[index]])));
  }
  throw new Error('Not static literal: ' + node.type);
}
function visit(node, callback) {
  if (!node || typeof node !== 'object') return;
  if (node.type) callback(node);
  for (const [key, child] of Object.entries(node)) if (!['start', 'end', 'loc'].includes(key)) {
    if (Array.isArray(child)) child.forEach(item => visit(item, callback)); else if (child && typeof child === 'object') visit(child, callback);
  }
}
function canonical(node, parameters = {}, expand = false, removeTargetedImport = false, propertyKey = false) {
  if (!node || typeof node !== 'object') return node;
  if (Array.isArray(node)) return node.map(child => canonical(child, parameters, expand, removeTargetedImport)).filter(child => child !== null);
  if (propertyKey && ['Identifier', 'Literal'].includes(node.type)) return { key: node.type === 'Identifier' ? node.name : String(node.value) };
  if (node.type === 'Identifier' && Object.hasOwn(parameters, node.name)) return canonical(parameters[node.name]);
  if (expand && node.type === 'CallExpression' && node.callee.type === 'Identifier' && node.callee.name === 'targetedHostedRuleSnapshot') {
    const fn = functions.targetedHostedRuleSnapshot;
    return canonical(fn.body.body[0].argument, Object.fromEntries(fn.params.map((parameter, index) => [parameter.name, node.arguments[index]])), false, false);
  }
  if (removeTargetedImport && node.type === 'ImportDeclaration' && node.source.value === './hosted-rule-snapshot.js') {
    const specifiers = node.specifiers.filter(specifier => specifier.imported?.name !== 'targetedHostedRuleSnapshot');
    return specifiers.length ? canonical({ ...node, specifiers }, parameters, expand, false) : null;
  }
  const normalized = {};
  for (const [key, child] of Object.entries(node)) {
    if (['start', 'end', 'loc', 'raw'].includes(key)) continue;
    normalized[key] = node.type === 'Property' && key === 'shorthand' ? false : canonical(child, parameters, expand, removeTargetedImport, node.type === 'Property' && key === 'key');
  }
  return normalized;
}
function uniqueObjects(root) {
  const set = new Set();
  function walk(value) { if (!value || typeof value !== 'object') return; assert.equal(set.has(value), false); set.add(value); Object.values(value).forEach(walk); }
  walk(root); return set.size;
}
