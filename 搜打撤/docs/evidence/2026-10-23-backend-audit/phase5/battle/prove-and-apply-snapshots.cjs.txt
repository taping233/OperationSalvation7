// A literal AST interpreter: no business module, migration, or helper is executed.
function value(node, parameters = {}, allowFactory = false) {
  if (node.type === 'Literal') return node.value;
  if (node.type === 'Identifier' && Object.hasOwn(parameters, node.name)) return parameters[node.name];
  if (node.type === 'UnaryExpression' && node.operator === '-') return -value(node.argument, parameters, allowFactory);
  if (node.type === 'ArrayExpression') return node.elements.map(element => value(element, parameters, allowFactory));
  if (node.type === 'ObjectExpression') {
    const object = {};
    for (const property of node.properties) {
      assert.equal(property.type, 'Property'); assert.equal(property.computed, false); assert.equal(property.kind, 'init'); assert.equal(property.method, false);
      const key = property.key.type === 'Identifier' ? property.key.name : property.key.value;
      assert.equal(Object.hasOwn(object, key), false, 'No duplicate snapshot property');
      Object.defineProperty(object, key, { value: value(property.value, parameters, allowFactory), writable: true, enumerable: true, configurable: true });
    }
    return object;
  }
  if (allowFactory && node.type === 'CallExpression') {
    assert.equal(node.callee.type, 'Identifier'); assert.equal(node.callee.name, 'hostedRuleSnapshot'); assert.equal(node.arguments.length, 1);
    const name = value(node.arguments[0]); assert.equal(typeof name, 'string');
    return value(helperNode.body.body[0].argument, { name });
  }
  throw new Error('Nonliteral or unsupported AST ' + node.type);
}

function allUniqueObjects(object) {
  const seen = new Set();
  function visit(value) {
    if (!value || typeof value !== 'object') return;
    assert.equal(seen.has(value), false, 'Every snapshot subtree must own its objects'); seen.add(value);
    for (const key of Object.keys(value)) visit(value[key]);
  }
  visit(object); return seen.size;
}
